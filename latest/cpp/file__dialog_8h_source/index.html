// SPDX-License-Identifier: BSD-3-Clause
#pragma once

#include <cstdint>
#include <filesystem>
#include <functional>
#include <optional>
#include <string>
#include <vector>

namespace ernie::app {

struct FileFilter {
    std::string label;
    std::vector<std::string> extensions;  

    bool Matches(const std::filesystem::path& path) const;
};

struct FileDialogOptions {
    std::string title = "Open";  
    bool save = false;
    std::vector<FileFilter> filters;
    std::filesystem::path start;
};

struct DirectoryEntry {
    std::string name;
    bool directory = false;
    std::uintmax_t size = 0;  
    std::filesystem::file_time_type modified{};
};

std::vector<DirectoryEntry> ListDirectory(const std::filesystem::path& directory,
                                          const FileFilter* filter, bool showHidden);

std::optional<std::filesystem::path> ChosenPath(const std::filesystem::path& directory,
                                                const std::string& name, const FileFilter* filter,
                                                bool save);

class FileDialog {
  public:
    using Done = std::function<void(const std::filesystem::path&)>;

    void Open(FileDialogOptions options, Done done);

    bool IsOpen() const {
        return open_;
    }

    const FileDialogOptions& Options() const {
        return options_;
    }

    const std::filesystem::path& Directory() const {
        return directory_;
    }

    bool Navigate(const std::filesystem::path& directory);

    bool Up();

    const std::string& Name() const {
        return name_;
    }

    void SetName(std::string name) {
        name_ = std::move(name);
    }

    std::size_t Filter() const {
        return filter_;
    }

    void SetFilter(std::size_t index);

    bool ShowHidden() const {
        return showHidden_;
    }

    void SetShowHidden(bool show) {
        showHidden_ = show;
    }

    const std::vector<DirectoryEntry>& Entries() const {
        return entries_;
    }

    void Refresh();

    bool Accept(const std::string& name = {});

    const std::optional<std::filesystem::path>& ReplacePending() const {
        return replace_;
    }

    bool ConfirmReplace();

    void Cancel();

    const std::string& Error() const {
        return error_;
    }

    const std::vector<std::filesystem::path>& Recent() const {
        return recent_;
    }

    void Draw();

  private:
    const FileFilter* ChosenFilter() const;
    void Choose(std::filesystem::path path);

    FileDialogOptions options_;
    Done done_;
    bool open_ = false;
    bool popupRequested_ = false;
    std::filesystem::path directory_;
    std::string pathField_;
    std::string name_;
    std::size_t filter_ = 0;
    bool showHidden_ = false;
    std::vector<DirectoryEntry> entries_;
    std::optional<std::filesystem::path> replace_;
    std::string error_;
    std::vector<std::filesystem::path> recent_;
    std::filesystem::path lastDirectory_;
};

}  // namespace ernie::app
