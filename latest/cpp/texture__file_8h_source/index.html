// SPDX-License-Identifier: BSD-3-Clause
#pragma once

#include "ernie/util/mapped_file.h"

#include <cstddef>
#include <cstdint>
#include <filesystem>
#include <memory>
#include <span>
#include <vector>

namespace ernie {

enum class TextureFormat : std::uint32_t {
    R8,
    Rg8,
    Rgba8,
    Rgba8Srgb,
    Bgra8,
    Bgra8Srgb,
    Rgba16F,
    Rgba32F,
    Bc1,  
    Bc1Srgb,
    Bc2,
    Bc2Srgb,
    Bc3,
    Bc3Srgb,
    Bc4,
    Bc4Snorm,
    Bc5,
    Bc5Snorm,
    Bc6hUfloat,
    Bc6hSfloat,
    Bc7,
    Bc7Srgb,
};

struct TextureFormatInfo {
    std::uint32_t blockSize = 1;   
    std::uint32_t blockBytes = 4;  
    std::uint32_t channels = 4;    
    bool srgb = false;             
    bool compressed() const {
        return blockSize > 1;
    }
};

TextureFormatInfo InfoOf(TextureFormat format);

TextureFormat WithSrgb(TextureFormat format, bool srgb);

std::size_t ImageBytes(TextureFormat format, std::uint32_t width,
                       std::uint32_t height);

struct TextureData {
    TextureFormat format = TextureFormat::Rgba8;
    std::uint32_t width = 0;
    std::uint32_t height = 0;
    std::uint32_t layers = 1;
    std::uint32_t mips = 1;
    bool cube = false;
    std::vector<std::byte> bytes;
    std::shared_ptr<const MappedFile> mapping;
    std::span<const std::byte> mapped;
    std::vector<std::size_t> offsets;

    std::span<const std::byte> Data() const {
        return mapping != nullptr ? mapped
                                  : std::span<const std::byte>(bytes);
    }

    std::span<const std::byte> Mip(std::uint32_t layer,
                                   std::uint32_t mip) const;
};

bool IsTextureFile(const std::filesystem::path& path);

TextureData LoadTextureFile(const std::filesystem::path& path);

TextureData ReadDds(std::span<const std::byte> file);
TextureData ReadKtx2(std::span<const std::byte> file);

TextureData DecodeBlocks(const TextureData& texture);

}  // namespace ernie
