enum class Dirty : std::uint32_t {
    None = 0,
    Added = 1u << 0,    
    Removed = 1u << 1,  
    Transform = 1u << 2,  
    Topology =
        1u
        << 3,  
    Points = 1u << 4,  
    Primvars = 1u << 5,    
    Material = 1u << 6,    
    Params = 1u << 7,      
    Visibility = 1u << 8,  
};

inline constexpr Dirty operator|(Dirty a, Dirty b) {
    return static_cast<Dirty>(static_cast<std::uint32_t>(a) |
                              static_cast<std::uint32_t>(b));
}

inline constexpr Dirty operator&(Dirty a, Dirty b) {
    return static_cast<Dirty>(static_cast<std::uint32_t>(a) &
                              static_cast<std::uint32_t>(b));
}

inline constexpr Dirty& operator|=(Dirty& a, Dirty b) {
    return a = a | b;
}

inline constexpr bool Any(Dirty bits, Dirty flags) {
    return (bits & flags) != Dirty::None;
}

Dirty DirtyOf(std::string_view attribute);

class PrimSet {
  public:
    PrimSet() = default;
    explicit PrimSet(std::vector<Path> paths)
        : paths_(std::move(paths)) {}

    const std::vector<Path>& Paths() const {
        return paths_;
    }

    std::size_t Size() const {
        return paths_.size();
    }

  private:
    friend class Scene;
    std::vector<Path> paths_;
    // The scene's prims as last resolved, valid for that scene
    // (`Scene::Id`) until it removes prims.
    mutable std::vector<void*> prims_;
    mutable std::uint64_t sceneId_ = 0;
    mutable std::uint64_t removals_ = 0;
    // Its own, for the columns scenes keep of it (`Scene::Write`);
    // copies share it, as they share its prims. 0 until first used.
    mutable std::uint64_t id_ = 0;
};

template <typename T>
T Prim::Get(std::string_view attribute, T fallback) const {
    return scene_ != nullptr
               ? scene_->Get<T>(path_, attribute, fallback)
               : fallback;
}
