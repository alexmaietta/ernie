enum class Dirty : std::uint32_t {
    None = 0,
    Added = 1u << 0,    
    Removed = 1u << 1,  
    Transform = 1u << 2,  
    Topology =
        1u
        << 3,  
    Points = 1u << 4,  
    Primvars = 1u << 5,    
    Material = 1u << 6,    
    Params = 1u << 7,      
    Visibility = 1u << 8,  
};

inline constexpr Dirty operator|(Dirty a, Dirty b) {
    return static_cast<Dirty>(static_cast<std::uint32_t>(a) |
                              static_cast<std::uint32_t>(b));
}

inline constexpr Dirty operator&(Dirty a, Dirty b) {
    return static_cast<Dirty>(static_cast<std::uint32_t>(a) &
                              static_cast<std::uint32_t>(b));
}

inline constexpr Dirty& operator|=(Dirty& a, Dirty b) {
    return a = a | b;
}

inline constexpr bool Any(Dirty bits, Dirty flags) {
    return (bits & flags) != Dirty::None;
}

Dirty DirtyOf(std::string_view attribute);

struct Change {
    Path path;
    Dirty bits = Dirty::None;
};

    std::uint64_t version = 0;

class Prim {
  public:
    Prim() = default;
    Prim(Scene* scene, Path path)
        : scene_(scene), path_(std::move(path)) {}

    bool IsValid() const;

    explicit operator bool() const {
        return IsValid();
    }

    const Path& GetPath() const {
        return path_;
    }

    const std::string& Type() const;

    Prim Define(std::string_view relative,
                std::string_view type) const;

    Prim GetPrim(std::string_view relative) const;

    Prim& Set(std::string_view attribute, Value value);

    Prim& Write(std::string_view attribute, Tensor tensor);

    Prim& SetXform(const XformOps& xform);

    const Value* Get(std::string_view attribute) const;

    template <typename T>
    T Get(std::string_view attribute, T fallback) const;

    const Tensor* Read(std::string_view attribute) const;

    std::vector<std::string> Attributes() const;

  private:
    Scene* scene_ = nullptr;
    Path path_;
};

class PrimSet {
  public:
    PrimSet() = default;
    explicit PrimSet(std::vector<Path> paths)
        : paths_(std::move(paths)) {}

    const std::vector<Path>& Paths() const {
        return paths_;
    }

    std::size_t Size() const {
        return paths_.size();
    }

  private:
    friend class Scene;
    std::vector<Path> paths_;
    // The scene's prims as last resolved, valid for that scene
    // (`Scene::Id`) until it removes prims.
    mutable std::vector<void*> prims_;
    mutable std::uint64_t sceneId_ = 0;
    mutable std::uint64_t removals_ = 0;
};

class Scene {
  public:

    std::vector<Path> Prims(std::string_view type = {}) const;

    void Set(const Path& path, std::string_view attribute,

    const Value* Get(const Path& path,

        const T* typed = value != nullptr ? value->Get<T>() : nullptr;

    struct UniqueId {
        std::uint64_t value;

        UniqueId();
        UniqueId(const UniqueId&) : UniqueId() {}
        UniqueId& operator=(const UniqueId&) {
            *this = UniqueId();
            return *this;
        }
        UniqueId(UniqueId&&) noexcept = default;
        UniqueId& operator=(UniqueId&&) noexcept = default;
    };

    struct PrimData {
        std::string type;
        const PrimTypeDef* schema = nullptr;
        std::map<std::string, Value, std::less<>> values;
        std::map<std::string, Tensor, std::less<>> tensors;
        std::map<std::string, std::uintptr_t, std::less<>>
            streams;  
        std::map<std::string, std::pair<Tensor, std::int64_t>,
                 std::less<>>
            rows;
        std::map<std::string, std::uint64_t, std::less<>> versions;
        std::array<std::uint64_t, kStampCount> stamps{};
        std::uint64_t ownTransform = 0;  
        std::uint64_t latest =
            0;  
    };

    PrimData& Find(const Path& path);
    const PrimData& Find(const Path& path) const;
    void Stamp(const Path& path, PrimData& prim,
               std::string_view attribute);
    void Stamp(const Path& path, PrimData& prim, Dirty bit);

                     bool checked = false);

    std::map<Path, PrimData> prims_;
    std::uint64_t removals_ = 0;  
    std::map<Path, std::uint64_t> removed_;
    std::set<std::pair<std::uint64_t, Path>> stamped_;
    std::map<std::string, std::set<Path>, std::less<>> byType_;
    std::map<std::string, Overlay, std::less<>> overlays_;
    std::uint64_t version_ = 0;
    UniqueId id_;
};

template <typename T>
T Prim::Get(std::string_view attribute, T fallback) const {
    return scene_ != nullptr
               ? scene_->Get<T>(path_, attribute, fallback)
               : fallback;
}
