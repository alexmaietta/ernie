// SPDX-License-Identifier: BSD-3-Clause
#pragma once

#include <cstddef>
#include <cstdint>
#include <span>
#include <vector>

namespace ernie {

struct MeshData {
    std::vector<float> points;  
    std::vector<std::uint32_t>
        indices;  
    std::vector<float> normals;  
    std::vector<float> cornerNormals;
    std::vector<float>
        colors;  
    std::vector<float> cornerColors;
    std::vector<float> uvs;

    struct Level {
        std::uint32_t first = 0;
        std::uint32_t count = 0;
        float error = 0.0f;
    };
    std::vector<Level> levels;
    std::vector<std::uint32_t> lodIndices;
    std::vector<float> lodUvs;
    std::vector<float> lodCornerNormals;
    std::vector<float> lodCornerColors;
};

struct TriangleCorners {
    std::vector<std::uint32_t> faces;
    std::vector<std::uint32_t> faceVertices;
};

TriangleCorners
TriangulatedCorners(std::span<const std::int64_t> faceVertexCounts);

std::vector<std::uint32_t>
TriangulateFaces(std::span<const std::int64_t> faceVertexCounts,
                 std::span<const std::int64_t> faceVertexIndices,
                 std::size_t pointCount);

struct InstanceData {
    std::vector<float> positions;  
    std::vector<float>
        orientations;  
    std::vector<float> scales;  
    std::vector<float> colors;  
    std::vector<float> temperatures;
    std::vector<float> visible;
};

}  // namespace ernie
