// SPDX-License-Identifier: BSD-3-Clause
#pragma once

#include <volk.h>

#include <cstdint>
#include <deque>
#include <mutex>
#include <unordered_map>
#include <vector>

namespace ernie {

class VulkanContext;

class BindlessHeap {
  public:
    enum class Kind : std::uint32_t {
        Texture2D = 0,           
        Texture2DArray = 1,      
        Sampler = 2,             
        ComparisonSampler = 3,   
        UintTexture2DArray = 4,  
    };
    static constexpr std::uint32_t kKinds = 5;
    static constexpr std::uint32_t kTextures = 4096;
    static constexpr std::uint32_t kSamplers = 256;
    static constexpr std::uint64_t kRetireFrames = 3;

    class Slot {
      public:
        Slot() = default;
        ~Slot();
        Slot(Slot&& other) noexcept;
        Slot& operator=(Slot&& other) noexcept;
        Slot(const Slot&) = delete;
        Slot& operator=(const Slot&) = delete;

        std::uint32_t Index() const {
            return index_;
        }
        bool Valid() const {
            return heap_ != nullptr;
        }

      private:
        friend class BindlessHeap;
        Slot(BindlessHeap* heap, Kind kind, std::uint32_t index)
            : heap_(heap), kind_(kind), index_(index) {}
        void Free();

        BindlessHeap* heap_ = nullptr;
        Kind kind_ = Kind::Texture2D;
        std::uint32_t index_ = 0;
    };

    explicit BindlessHeap(const VulkanContext& context);
    ~BindlessHeap();
    BindlessHeap(const BindlessHeap&) = delete;
    BindlessHeap& operator=(const BindlessHeap&) = delete;

    Slot AddImage(Kind kind, VkImageView view,
                  VkImageLayout layout =
                      VK_IMAGE_LAYOUT_SHADER_READ_ONLY_OPTIMAL);

    Slot AddSampler(VkSampler sampler, bool comparison = false);

    void Forget(std::uint64_t object);

    void AdvanceFrame();

    VkDescriptorSetLayout Layout() const {
        return layout_;
    }

    void Bind(VkCommandBuffer commands, VkPipelineBindPoint point,
              VkPipelineLayout pipelineLayout) const;

    std::uint32_t InUse(Kind kind) const;

  private:
    struct Array {
        std::uint32_t capacity = 0;
        std::uint32_t next = 0;  
        std::vector<std::uint32_t> free;
        std::deque<std::pair<std::uint32_t, std::uint64_t>> retiring;
        std::uint32_t inUse = 0;
    };

    std::uint32_t Take(Kind kind);
    void Release(Kind kind, std::uint32_t index);
    void Write(Kind kind, std::uint32_t index,
               const VkDescriptorImageInfo& info);

    const VulkanContext& context_;
    VkDescriptorSetLayout layout_ = VK_NULL_HANDLE;
    VkDescriptorPool pool_ = VK_NULL_HANDLE;
    VkDescriptorSet set_ = VK_NULL_HANDLE;
    mutable std::mutex mutex_;
    Array arrays_[kKinds];
    struct Held {
        std::uint64_t object = 0;
        std::uint32_t layout = 0;
        bool operator==(const Held&) const = default;
    };
    struct HeldHash {
        std::size_t operator()(const Held& held) const {
            return std::hash<std::uint64_t>()(held.object) ^
                   (std::hash<std::uint32_t>()(held.layout) << 1);
        }
    };
    struct SharedSlot {
        Held held;
        std::uint32_t holders = 0;
    };
    std::unordered_map<Held, std::uint32_t, HeldHash> shared_[kKinds];
    std::unordered_map<std::uint32_t, SharedSlot> sharedAt_[kKinds];

    Slot Share(Kind kind, const Held& held,
               const VkDescriptorImageInfo& info);
    std::uint64_t frame_ = 0;
};

}  // namespace ernie
