struct DemoInfo {
    std::string category;     
    std::string name;         
    std::string description;  
    std::function<std::shared_ptr<Demo>(Viewer*, const DemoOptions&)>
        create;

    std::string Label() const;

    std::string Id() const;
};

template <typename T>
DemoInfo MakeDemoInfo(std::string category, std::string name,
                      std::string description = {}) {
    return {std::move(category), std::move(name),
            std::move(description),
            [](Viewer* viewer,
               const DemoOptions& options) -> std::shared_ptr<Demo> {
                return std::make_shared<T>(viewer, options);
            }};
}

    static const ::ernie::app::DemoRegistration                      \
        ernie_demo_registration_##Class(                             \
            ::ernie::app::MakeDemoInfo<Class>(category, name,        \
