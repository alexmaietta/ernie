// SPDX-License-Identifier: BSD-3-Clause
#pragma once

#include <volk.h>

#include <cstdint>
#include <mutex>
#include <vector>

namespace ernie {

class GpuTimer {
  public:
    GpuTimer(VkDevice device, std::uint32_t timestampBits,
             float periodNs);
    ~GpuTimer();
    GpuTimer(const GpuTimer&) = delete;
    GpuTimer& operator=(const GpuTimer&) = delete;

    std::int64_t Begin(VkCommandBuffer commands, const char* name);
    void End(VkCommandBuffer commands, std::int64_t zone);

    void Collect();

  private:
    struct Pending {
        const char* name;
        std::uint32_t query;  
        std::uint64_t frame;
        std::uint32_t depth;
        bool ended = false;
    };

    struct Retiring {
        std::uint32_t query;
        std::uint32_t collects = 0;
    };

    static constexpr std::uint32_t kQueries = 4096;
    static constexpr std::uint32_t kRetireCollects = 4;

    VkDevice device_;
    VkQueryPool pool_ = VK_NULL_HANDLE;
    std::uint64_t mask_ = 0;
    double periodNs_ = 1.0;
    std::mutex mutex_;
    std::vector<Pending> pending_;
    std::vector<Retiring> retiring_;
    std::vector<std::uint32_t>
        free_;  
    std::uint32_t depth_ =
        0;  
};

}  // namespace ernie
